import pandas as pd

df = pd.read_csv('master_1to3.csv', encoding='utf-8')
# 기본 집계: 시군구별 전체
agg_region = df.groupby(['region_code','region_name']).agg(
    total_generated = ('generated_kg','sum'),
    total_recycled = ('recycled_kg','sum'),
    imputed_count = ('imputed_flag', lambda x: x.astype(bool).sum()),
    record_count = ('imputed_flag', 'count')
).reset_index()
agg_region['recycling_rate'] = agg_region['total_recycled'] / agg_region['total_generated']
agg_region['imputed_ratio'] = agg_region['imputed_count'] / agg_region['record_count']

# 하위 5개(재활용률 기준)
vulnerable_by_rate = agg_region.sort_values('recycling_rate').head(5)

# imputed 높은 상위 10개
vulnerable_by_imputed = agg_region.sort_values('imputed_ratio', ascending=False).head(10)

agg_comb = df.groupby(['region_code','region_name','material_code']).agg(
    total_generated = ('generated_kg','sum'),
    total_recycled = ('recycled_kg','sum'),
    imputed_count = ('imputed_flag', lambda x: x.astype(bool).sum()),
    record_count = ('imputed_flag','count')
).reset_index()
agg_comb['recycling_rate'] = agg_comb['total_recycled'] / agg_comb['total_generated']
agg_comb['imputed_ratio'] = agg_comb['imputed_count'] / agg_comb['record_count']
vulnerable_comb = agg_comb.sort_values('recycling_rate').query('total_generated>0').head(20)
